function esc(s){return String(s).replace(/[&<>"']/g,function(m){return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m];});}

function choiceRow(label,count,denominator){
  return '<div class="choice"><span>'+esc(label)+'</span><span class="count">'+count+'</span><div class="bar"><span style="width:'+Math.round(count/denominator*100)+'%"></span></div></div>';
}

function renderQuestionCard(q,overall){
  var inner='<h3>'+esc(q.prompt)+'</h3><div class="denominator">有效分母：'+esc(q.denominator)+'</div>';
  if(q.type==='scale' || q.type==='number'){
    inner+='<div>均值 '+esc(q.mean)+' · 中位数 '+esc(q.median)+' '+esc(q.unit||'')+'</div>';
  }
  if(q.choices && q.choices.length){
    q.choices.forEach(function(c){
      inner+=choiceRow(c.label,c.count,q.denominator);
    });
  }
  return inner;
}

function renderOverall(){
  var html='';
  QUESTIONS.forEach(function(q){
    var attr='data-question-id="'+esc(q.id)+'"';
    html+='<div class="question" '+attr+'>'+renderQuestionCard(q,true)+'</div>';
  });
  document.getElementById('overall-stats').innerHTML=html;
}

function renderGroup(){
  var summary=document.getElementById('group-summary');
  var stats=document.getElementById('group-stats');
  var groupName=ORDER[idx];
  var valid=groupName==='全部'?12:4;
  summary.innerHTML='<span class="group-name">'+esc(groupName)+'</span>（有效数 <span class="group-valid">'+valid+'</span>）';

  if(groupName==='全部'){
    stats.innerHTML='<p>当前为全部分组，整体统计已在上方 #overall-stats 显示。</p>';
    return;
  }

  var group=GROUP_DATA[groupName];
  var html='';
  QUESTIONS.forEach(function(q){
    if(!group.hasOwnProperty(q.id)){return;}
    var token=group[q.id];
    var inner='<h3>'+esc(q.prompt)+'</h3><div class="denominator">有效分母：4</div>';
    if(q.id==='privacy' || q.id==='max-monthly-payment'){
      inner+='<div>均值 '+esc(token.mean)+' · 中位数 '+esc(token.median)+' '+esc(q.unit||'')+'</div>';
    }else if(q.choices && token.choices){
      token.choices.forEach(function(count,i){
        inner+=choiceRow(q.choices[i].label,count,4);
      });
    }
    html+='<div class="question" data-group-question-id="'+esc(q.id)+'">'+inner+'</div>';
  });
  stats.innerHTML=html;
}
